```bash
  --records-json artifacts/paper_draft_budget_plus_high/records.json \
  --output-root artifacts/paper_draft_budget_plus_high/figures \
  --status draft

  --records-json artifacts/paper_draft_budget_plus_high/records.json \
  --summary-json artifacts/paper_draft_budget_plus_high/derived/summary.json \
  --output-root artifacts/paper_draft_budget_plus_high/tables \

  artifacts/paper_draft_budget_plus_high \
  --summary-json artifacts/paper_draft_budget_plus_high/derived/summary.json
```

```bash

  artifacts/paper_draft_budget_plus_high \

  --workload-manifest artifacts/paper_draft_gpu_combined/configs/workloads.json \
```
